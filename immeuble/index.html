Code de l'immeuble

from turtle import *
import random as rd

j_n = rd.randint(0,1)

def environment(j_n):
    """Definie un environnement de jour ou de nuit."""
    up()
    goto(-700, -225)
    down()

    if j_n == 1:
        pencolor("#1A1A40")
        fillcolor("#1A1A40")  # Couleur de nuit

    else:
        pencolor("#C0E8F9")
        fillcolor("#C0E8F9")  # Couleur de jour

    begin_fill()
    speed(0)
    for i in range(4):
        fd(1350)
        lt(90)
    end_fill()

    pencolor("green")
    fillcolor("green")
    begin_fill()
    for i in range(4):
        fd(1350)
        rt(90)
    end_fill()

    return j_n


def couleur():
    return (rd.random(), rd.random(), rd.random())

def etage(x,y):
    """
    créer un étage
    """
    goto(x,y)
    down()
    for i in range(2):
        forward(140)
        left(90)
        forward(60)
        left(90)
    up()

def fenetre(x,y,j_n):
    goto(x,y)
    down()
    if j_n == 1:
        fillcolor("gold")  # Couleur de nuit

    else:
        fillcolor("#5C7490") 
    begin_fill()
    for j in range(3):
        for i in range(4):
            forward(30)
            left(90)
        x += 40
        up()
        goto(x,y)
        down()
    end_fill()
    up()

def toit(x,y):
    goto(x-5,y)
    down()
    fillcolor("#54475e")
    begin_fill()
    choix = rd.randint(0,1)
    if choix == 0:
        fd(150)
        lt(150)
        fd(88)
        lt(60)
        fd(88)
        lt(150)
    else:
        for i in range(2):
            fd(150)
            lt(90)
            fd(15)
            lt(90)
    end_fill()
    up()

def porte(x,y):
    x = x + 40 * rd.randint(0,2)
    goto(x,y)
    down()
    c = couleur()
    fillcolor(c)
    begin_fill()
    choix = rd.randint(0,1)
    if choix == 0:
        for i in range(2):
            fd(30)
            lt(90)
            fd(50)
            lt(90)
    else:
        fd(30)
        lt(90)
        fd(40)
        circle(15,180)
        fd(40)
        lt(90)
    end_fill()
    up()

def balcon(x,y,j_n):
    x = x + 40 * rd.randint(0,2)
    goto(x,y)
    down()
    if j_n == 1:
        fillcolor("gold")  # Couleur de nuit

    else:
        fillcolor("#5C7490") 

    begin_fill()
    for i in range(2):
        fd(30)
        lt(90)
        fd(45)
        lt(90)
    end_fill()
    x -= 5
    up()
    goto(x,y)
    down()
    for i in range (8):
        fd(5)
        lt(90)
        fd(20)
        lt(90)
        fd(5)
        lt(90)
        fd(20)
        lt(90)
        fd(5)
    up()


def immeuble_1(j_n):
    x = -500
    y = -250
    up()
    goto(x,y)
    compteur = 0
    fillcolor(couleur())
    begin_fill()
    for i in range(rd.randint(1,5)):
        etage(x,y)
        y += 60
        compteur += 1
    end_fill()
    toit(x,y)
    x += 15
    y = y - 60*compteur + 15
    for i in range(compteur):
        fenetre(x,y,j_n)
        y += 60
    y = y - 60*compteur - 15
    porte(x,y)
    for i in range(compteur-1):
        x = -485
        y += 60
        balcon(x,y,j_n)

Explication du code

Explication du code partie par partie:

Programme secondaire:

  • Fonction environement:

Il s'agit d'une fonction qui a pour but de définir si l'environnement sera présenté de jour ou de nuit, grace a une valeur donné à j_n, dans le cas ou elle est égale à 0 on aura un environnement de jour et s'il est égale à 1 alors on aura un environnement de nuit, il déssinera aussi de l'herbe dans l'environnement.

  • Fonction couleur:

Il s'agit d'une fonction qui a pour but de créer une une couleur aléatoire lors de l'appelle de la fonction en insérant un nombre aléatoire dans chaque valeur de RVB.

  • Fonction etage:

Il s'agit d'une fonction qui a pour but de créer un rectangle de taille 140 * 60 pixel au coordonné x,y donné.

  • Fontion fenetre:

Il s'agit d'une fonction qui a pour but de créer 3 carré de 30 * 30 pixel chacun espacé de 10 pixel et qui les remplient d'une couleur définie en fonction du résultat de j_n.

  • Fonction toit:

Il s'agit d'une fonction qui a pour but de créer des toits, un en forme de triangle isocèlle et l'autre en forme de rectangle de 150 * 15 pixel, choisis aléatoirement grace a rd.randint.

  • Fontion porte:

Il s'agit d'une fonction qui a pour but de créer des portes, une rectagulaire et l'autre rectangulaire avec un haut arondie, choisis aléatoirement grace a rd.randint et remplient d'une couleur aléatoire grace à la fonction couleur.

  • Fonction balcon:

Il s'agit d'une fonction qui a pour but de créer un balcon à la place d'une des 3 fenêtres.

  • Fontion immeuble:

Il s'agit d'une fonction qui a pour but de créer un immeuble a des coordonné x,y donnés en utilisant toutes les fonctions cité dans l'ordre : etage répéter de une à 5 fois aléatoirement, toit situé en haut de l'immeuble, fenetre qui créer 3 fenêtres par étage, porte qui créer une porte étage 0 à la place d'une des 3 fenêtres de magnière aléatoire, balcon qui créer un bacon à la place d'une des 3 fenêtres aléatoirement à chaque étage.

  • Fonction ville:

La fonction pricipale qui va créer toute notre ville en appelant environnement pour définir le décore, et les fonctions immeuble pour créer nos 5 immeubles et former une ville.

Programme Principale:

Enfin on définie la vittesse de la tortue à 0 car il s'agit de la vitesse la plus rapide, on appelle la fontion ville et on écrit done() pour utiliser turtle.
