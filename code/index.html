import random as random

lettre = "X "  #lettre pour faire les cotés
espace = "  "  #espace pour l'intérieur du carré
def carre(cote):
   """premier carré avec des cotés de 8"""
   print(lettre * cote)  #cela fait le dessus du carré 
   for i in range(cote - 2):  
      print(lettre + espace * 6 + lettre)  #cela fais le coté gauche et le coté droite du carré
   print(lettre * cote)  #cela fait le dessous du carré
   return

print(carre(8))  #cela fait le carré avec des cotés de 8 "X"

def carre2(cote2):
   """deuxième carré avec des cotés de 9 et une croix à l'intérieur"""
   print(lettre * cote2)  #cela fait le dessus du carré
   for i in range(cote2 - 6):
      print(lettre + espace * 3 + lettre + espace * 3 + lettre)  #cela fait la debut du coté gauche, du coté droite et de la croix
   print(lettre * cote2)  #cela fait la barre du milieu pour la croix
   for i in range(cote2 - 6):
      print(lettre + espace * 3 + lettre + espace * 3 + lettre)  #cela fait la fin du coté gauche, du coté droite et de la croix
   print(lettre * cote2)  #cela fais le dessous du carré
   return

print(carre2(9))  #cela fais le carré avec des cotés de 9 "X" et la croix au milieu

def carre3(cote2):
   """troixième carré avec une barre en diagonal à l'intérieur"""
   a = 0 # 1er espace
   a1 = 6 # 2eme espace
   print(lettre * cote2)  #cela fait le dessus du carré
   for i in range(cote2 - 2):
      print(lettre + espace * a + lettre + espace * a1 + lettre)  #cela fait le coté gauche, le coté droit et la barre en diagonal
      a += 1  #ajoute 1 au première espace
      a1 -= 1  #retire 1 au deuxième espace
   print(lettre * cote2)  #cela fait le dessous du carré
   return

print(carre3(9))  #cela fait le carré avec une barre en diagonal à l'intérieur

def carre4(cote2):
   a = 6 # 1er espace
   a1 = 0 # 2eme espace
   a2 = 1 # 3eme espace
   print(lettre * cote2) # cela fait le dessus du carré
   for i in range(cote2 - cote2//2 - 1):
      print(lettre + espace * a + lettre + espace * a1 + lettre)  #cela fait le debut des cotés droit et gauche du carré et le debut de la digonal jusqu'au milieu
      a -= 1  #retire 1 au première espace
      a1 += 1 #ajoute 1 au deuxième espace
   for i in range(cote2 - cote2//2 - 2): 
      a1 -= 1 #retire 1 au première espace
      print(lettre + espace * a + lettre + espace * a2 + lettre + espace * (a1-1) + lettre) #cela fait le reste des coté droit et gauche et cela fait aussi un triangle a partir du milieu dans le carré
      a -= 1 #retire 1 au première espace
      a2 += 2 #ajoute 1 au troixième espace
   print(lettre * cote2)  #cela fait le dessous du carré
   return

print(carre4(9)) #cela fait le carré avec une digonal jusqu'au milieu puis un triangle à l'intérieur du carré

hauteur = 5  #définie la hauteur du triangle

def triangle(hauteur):
   b = 5 #1er espace
   b1 = 1 #2eme espace
   print(espace * b + lettre) #cela fait le dessus du triangle
   for i in range(hauteur-1): #cela fait les coté droit et gauche du triangle
      b -= 1 #retire 1 au premier espace
      print(espace * b + lettre + espace * b1 + lettre)
      b1 += 2 #ajoute 1 au deuxième espace
   print(lettre * 11) #cela fait le dessous du triangle
   return

print(triangle(hauteur)) #cela fait le triangle

def triangle2(hauteur1): #cela fait le même triangle que celui d'avant mais inversé
   b = 5 #1er espace
   b1 = 1 #2eme espace
   b2 = 5 #3eme espace
   print(lettre * 11) #cela fait la ligne du dessus du triangle
   print(lettre * b + espace + lettre * b2) #cela fait le dessus du tringle
   for i in range(hauteur-1): #cela fait les cotés droit et gauche du triangle
      b -= 1 #retire 1 au première espace
      b2 -= 1 #retire 2 au troixième espace
      print(lettre * b + espace + espace * b1 + espace + lettre * b2)
      b1 += 2 #ajoute 2 au deuxième espace
   print(lettre * 11) #cela fait la ligne en dessous du triangle
   return

print(triangle2(6)) #cela fait le tringle encadré par des lettres

espace1 = " "
def sapin(hauteur): # c'est le premier test de sapin sans les boules qui apparaissent aléatoirement
   e = hauteur
   e1 = 0
   print(espace1 * e + "*")
   print(espace1 * e + "^")
   for i in range(hauteur):
      e -= 1
      print(espace1 * e + "/" + "\"" + ("\'" + "\"") * e1 + "\\")
      e1 += 1
   for i in range(3):
      print(espace1 * (hauteur-1) + "|" * 3)
   return

print(sapin(7))

espace2 = " " #cela définie l'espace pour le sapin
def sapin1(hauteur):
   e = hauteur #1er espace
   print(espace2 * e + "*") #cela fait l'étoile du sapin
   print(espace2 * e + "^") #cela fait le dessus du sapin
   for i in range(hauteur): #cela fait tout le sapin avec des boules qui apparaisent aléatoirement
      lettre = ""
      for j in range(2 * i + 1): #cela fait tout l'intérieur du sapin
         if random.random() < 0.2: #cela rajoute une boule (20% de chance)
            lettre += "o"
         else:
            lettre += random.choice(["'", '"']) #cela rajoute ' ou " 
      e -= 1 #retire 1 au première espace
      print(espace2 * e + "/" + lettre + "\\") #cela fait les cotés et l'intérieur du sapin
   for i in range(3):
      print(espace2 * (hauteur-1) + "|" * 3) #cela fait le pied du sapin
   return

print(sapin1(7)) #cela fait le sapin avec des boules aléatoire
